        return false;
      }

      state.projectName=String(saved.projectName||'Новая программа');
      state.recipientName=String(saved.recipientName||'');
      state.coachName=String(saved.coachName||'');
      state.projectNote=String(saved.projectNote||'');
      state.folders=Array.isArray(saved.folders)?saved.folders:[];
      state.exercises=Array.isArray(saved.exercises)?saved.exercises:[];
      state.plans=Array.isArray(saved.plans)?saved.plans:[];
      state.schedule=new Map(Array.isArray(saved.schedule)?saved.schedule:[]);
      state.originalSchedule=new Map(Array.isArray(saved.originalSchedule)?saved.originalSchedule:[]);
      state.originalFolderKeys=new Set(Array.isArray(saved.originalFolderKeys)?saved.originalFolderKeys:[]);
      state.originalExerciseKeys=new Set(Array.isArray(saved.originalExerciseKeys)?saved.originalExerciseKeys:[]);
      state.originalPlanKeys=new Set(Array.isArray(saved.originalPlanKeys)?saved.originalPlanKeys:[]);
      state.selectedPlanKey=saved.selectedPlanKey || state.plans[0]?.key || null;

      const restoredCursor=saved.cursor?new Date(saved.cursor):startOfWeek(new Date());
      state.cursor=Number.isNaN(restoredCursor.getTime())?startOfWeek(new Date()):restoredCursor;
      state.analytics=desktopAnalytics&&Array.isArray(desktopAnalytics.workouts)?desktopAnalytics:null;

      setAutosaveState('ВОССТАНОВЛЕНО');
      return true;
    }catch(err){
      console.error('Workspace restore:',err);
      setAutosaveState('НЕ УДАЛОСЬ ВОССТАНОВИТЬ');
      return false;
    }
  }

  function deltaHtml(current,previous){
    const d=pctDelta(current,previous);
    if(d===null) return '<span class="delta-up">новая активность</span>';
    const cls=d>0.5?'delta-up':d<-0.5?'delta-down':'delta-flat';
    const sign=d>0?'+':'';
    return `<span class="${cls}">${sign}${d.toLocaleString('ru-RU',{maximumFractionDigits:1})}% к предыдущему периоду</span>`;
  }
