  function renderExercises(){

    if(!state.exercises.length){
      root.innerHTML='<div class="card"><div class="empty">Справочник пуст. Добавь первое упражнение.</div></div>';
      return;
    }

    const orderedFolders=[
      ...[...state.folders].sort((a,b)=>a.sortOrder-b.sortOrder||a.name.localeCompare(b.name,'ru')).map(f=>({key:f.key,name:f.name})),
      {key:null,name:'БЕЗ ПАПКИ'}

            </tr>`).join('')}</tbody>
        </table>
      </div>`;
    }).join('');

    root.querySelectorAll('input,select').forEach(el=>{
      el.addEventListener('change',()=>{
        const tr=el.closest('tr');
        const ex=state.exercises.find(x=>x.key===tr?.dataset.key);
        if(!ex) return;
        if(el.dataset.field==='factor') ex.factor=Number(el.value)===2?2:1;
        else if(el.dataset.field==='folderKey') ex.folderKey=el.value||null;
        else ex[el.dataset.field]=el.value.trim() || (el.dataset.field==='muscleGroup'?'Другое':'Без названия');
        renderAll();
        queueWorkspaceSave();
      });
    });

    root.querySelectorAll('[data-delete-exercise]').forEach(btn=>btn.addEventListener('click',()=>{
      const key=btn.closest('tr')?.dataset.key;
      const ex=state.exercises.find(x=>x.key===key);
      if(!confirm(`Списать «${ex?.name||'упражнение'}»? Оно будет удалено и из планов.`)) return;
      state.exercises=state.exercises.filter(x=>x.key!==key);
      state.plans.forEach(p=>p.items=p.items.filter(i=>i.exerciseKey!==key));
      renderAll();
      saveWorkspaceNow();
    }));
  }

        }).join('')}</tbody>
      </table>`:'<div class="empty">Сначала добавь упражнения.</div>'}`;
    $('#planNameInput')?.addEventListener('change',e=>{
      plan.name=e.target.value.trim()||'Без названия';
      renderPlanList();
      renderSchedule();
      queueWorkspaceSave();
    });
    $('#deletePlanBtn')?.addEventListener('click',()=>{
      if(!confirm(`Удалить план «${plan.name}»? Назначения этого плана будут сняты.`)) return;
      state.plans=state.plans.filter(p=>p.key!==plan.key);
      for(const [d,k] of [...state.schedule.entries()]) if(k===plan.key) state.schedule.delete(d);
      state.selectedPlanKey=state.plans[0]?.key||null;
      renderAll();
      saveWorkspaceNow();

      if(ch.checked){

      }
      renderPlans();
      queueWorkspaceSave();
    }));

      const item=plan.items.find(i=>i.exerciseKey===inp.dataset.sets);
      if(item){
        item.setCount=Math.max(1,Math.min(20,Number(inp.value)||3));
        queueWorkspaceSave();
      }
    }));
  }

  function renderSchedule(){
    const root=$('#scheduleTable');
    if(!state.plans.length){
      root.innerHTML='<div class="empty">Для графика нужен хотя бы один план.</div>'; return;
    }
    const days=Array.from({length:28},(_,i)=>addDays(state.cursor,i));
    root.innerHTML=`<table>
      <thead><tr><th>ДАТА</th><th>НАЗНАЧЕНИЕ</th></tr></thead>
      <tbody>${days.map(d=>{
        const k=dateKey(d), selected=state.schedule.get(k)||'';
        return `<tr>
          <td style="width:220px"><div class="date-main">${esc(fmtDate(d))}</div><div class="date-sub">${k}</div></td>
          <td><select data-schedule="${k}">
            <option value="">— отдых / без назначения —</option>
            ${state.plans.map(p=>`<option value="${esc(p.key)}"${selected===p.key?' selected':''}>${esc(p.name)}</option>`).join('')}
          </select></td>
        </tr>`;
      }).join('')}</tbody>
    </table>`;
    root.querySelectorAll('[data-schedule]').forEach(sel=>sel.addEventListener('change',()=>{
      const k=sel.dataset.schedule;
      if(sel.value) state.schedule.set(k,sel.value); else state.schedule.delete(k);
      queueWorkspaceSave();
    }));
  }

  function analyticsSlice(){
    const a=state.analytics;
    if(!a) return null;

    const now=new Date();
    const cutoff=rangeCutoff(state.statsRange,now);
    const workouts=a.workouts.filter(w=>!cutoff || safeDate(w.finishedAt)>=cutoff);
    const ids=new Set(workouts.map(w=>w.id));
    let sets=a.sets.filter(s=>ids.has(s.workoutId));

    if(state.selectedExercise!=='all'){
      sets=sets.filter(s=>exerciseKeyForSet(s)===state.selectedExercise);
      const setWorkoutIds=new Set(sets.map(s=>s.workoutId));
      return {a,now,cutoff,workouts:workouts.filter(w=>setWorkoutIds.has(w.id)),sets};
    }
    return {a,now,cutoff,workouts,sets};
  }
