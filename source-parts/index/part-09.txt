    if(trend.delta===null) return '<span class="delta-up">новая нагрузка против прошлых 4 недель</span>';

  }

  async function savePlanning(){
    const payload=currentPackage();
    const text=JSON.stringify(payload,null,2);
    const filename=planningFilename();
    if(window.desktopAPI){
      const saved=await window.desktopAPI.savePlanningFile(text,filename);
      if(!saved) return null;
      setStatus(`Программа сохранена: ${saved.name} · ${payload.data.exercises.length} упражнений · ${payload.data.plans.length} планов · ${payload.data.schedule.length} назначений.`,'ok');
      return saved;
    }
    downloadText(text,filename);
