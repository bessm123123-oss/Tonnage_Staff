  function fmtDecimal(value,digits=1){
    return (Number(value)||0).toLocaleString('ru-RU',{maximumFractionDigits:digits});
  }
  function fmtDuration(minutes){
    const value=Math.max(0,Number(minutes)||0);
    if(value<60) return `${Math.round(value)} мин`;
    const hours=Math.floor(value/60),mins=Math.round(value%60);
    return `${hours} ч${mins?` ${mins} мин`:''}`;
  }
  function metricCard(label,value,sub=''){
    return `<div class="metric"><div class="metric-label">${esc(label)}</div><div class="metric-value">${esc(value)}</div><div class="metric-sub">${sub}</div></div>`;
  }
  function statsHbars(rows,valueField='value',valueFormatter=fmtWeight){
    if(!rows.length) return '<div class="empty">Данных нет.</div>';
    const max=Math.max(1,...rows.map(row=>Number(row[valueField])||0));
    return `<div class="hbar-list">${rows.map(row=>{
      const value=Number(row[valueField])||0;
      return `<div class="hbar-row"><div class="hbar-name" title="${esc(row.name||row.key)}">${esc(row.name||row.key)}</div>
        <div class="hbar-track"><div class="hbar-fill" style="width:${Math.max(value?1:0,value/max*100)}%"></div></div>
        <div class="hbar-value">${esc(valueFormatter(value))}</div></div>`;
    }).join('')}</div>`;
  }
  function statsSeriesBars(rows,count=26){
    const entries=rows.slice(-count);
    if(!entries.length) return '<div class="empty">Данных нет.</div>';
    const max=Math.max(1,...entries.map(row=>row.value));
    return `<div class="chart slim">${entries.map(row=>`<div class="bar-wrap" title="${esc(row.key)} · ${esc(fmtWeight(row.value))}${row.rolling4!==undefined?` · средняя 4 недели ${esc(fmtWeight(row.rolling4))}`:''}">
      <div class="bar-value">${row.value?esc(fmtWeight(row.value)):''}</div><div class="bar" style="height:${Math.max(2,row.value/max*125)}px"></div>
      <div class="bar-label">${esc(fmtShortDate(row.key))}</div></div>`).join('')}</div>`;
  }
  function modelHeatmap(model){
    const end=startOfLocalDay(model.metrics.lastDate||new Date());
    const start=addDays(end,-90);
    const daily=new Map(model.daySeries.map(row=>[row.key,row.value]));
    const max=Math.max(1,...daily.values());
    const startMonday=startOfIsoWeek(start);
    const weeks=[];
    for(let wi=0;wi<13;wi++){
      const cells=[];
      for(let di=0;di<7;di++){
        const d=addDays(startMonday,wi*7+di),key=dateKey(d),value=daily.get(key)||0;
        const ratio=value/max;
        const level=!value?'':ratio>.72?'l4':ratio>.42?'l3':ratio>.18?'l2':'l1';
        cells.push(`<div class="heat-day ${level}" title="${key}${value?' · '+fmtWeight(value):''}"></div>`);
      }
      weeks.push(`<div class="heat-week">${cells.join('')}</div>`);
    }
    return `<div class="heatmap">${weeks.join('')}</div><div class="heat-legend">меньше <span class="legend-cell" style="background:#12161b"></span><span class="legend-cell" style="background:#2f343a"></span><span class="legend-cell" style="background:#555d65"></span><span class="legend-cell" style="background:#8f969e"></span><span class="legend-cell" style="background:#e4e8eb"></span> больше</div>`;
