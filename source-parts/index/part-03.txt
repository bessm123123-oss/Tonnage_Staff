    const el=document.querySelector('#autosaveState');
    if(el) el.textContent=text;
  }

  function saveWorkspaceNow(){
    try{
      const snapshot=workspaceSnapshot();
      localStorage.setItem(WORKSPACE_STORAGE_KEY,JSON.stringify(snapshot));
      window.desktopAPI?.saveWorkspace(snapshot).catch(err=>{
        console.error('Native workspace autosave:',err);
        setAutosaveState('ОШИБКА СОХРАНЕНИЯ');
      });
      setAutosaveState('СОХРАНЕНО ЛОКАЛЬНО');

    }catch(err){

      setAutosaveState('ОШИБКА СОХРАНЕНИЯ');
      return false;
    }
  }

  function queueWorkspaceSave(){
    setAutosaveState('СОХРАНЕНИЕ…');
